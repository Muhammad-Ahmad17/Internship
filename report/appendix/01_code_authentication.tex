\section{JWT Authentication Implementation}

\subsection{Express Middleware for Protected Routes}

\begin{lstlisting}[language=JavaScript, caption=JWT Verification Middleware]
// Verify and decode JWT token from request header
const verifyToken = (req, res, next) => {
  const token = req.headers['authorization']?.split(' ')[1];
  
  if (!token) {
    return res.status(403).json({ 
      message: 'No token provided' 
    });
  }
  
  jwt.verify(token, process.env.JWT_SECRET, (err, decoded) => {
    if (err) {
      return res.status(401).json({ 
        message: 'Invalid or expired token' 
      });
    }
    req.userId = decoded.id;
    next();
  });
};

// Apply to protected routes
app.get('/protected', verifyToken, (req, res) => {
  res.json({ message: 'Access granted', userId: req.userId });
});
\end{lstlisting}

\subsection{User Registration with Password Hashing}

\begin{lstlisting}[language=JavaScript, caption=Secure User Registration]
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');

const registerUser = async (req, res) => {
  try {
    const { email, password, name } = req.body;
    
    // Check if user exists
    const existingUser = await User.findOne({ email });
    if (existingUser) {
      return res.status(400).json({ 
        message: 'User already exists' 
      });
    }
    
    // Hash password with 10 salt rounds
    const hashedPassword = await bcrypt.hash(password, 10);
    
    // Create user
    const user = new User({
      name,
      email,
      password: hashedPassword
    });
    
    await user.save();
    
    // Generate JWT token
    const token = jwt.sign(
      { id: user._id, email: user.email },
      process.env.JWT_SECRET,
      { expiresIn: '24h' }
    );
    
    res.status(201).json({
      message: 'User registered successfully',
      token,
      user: { id: user._id, email: user.email }
    });
  } catch (error) {
    res.status(500).json({ message: error.message });
  }
};
\end{lstlisting}
