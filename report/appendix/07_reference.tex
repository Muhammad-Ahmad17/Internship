\section{Security Best Practices}

\subsection{Authentication and Authorization Checklist}

\begin{enumerate}
    \item \textbf{Password Storage:} Use bcrypt with minimum 10 salt rounds
    \item \textbf{JWT Tokens:} Include expiration time (recommended: 1 hour)
    \item \textbf{Token Refresh:} Implement refresh token rotation mechanism
    \item \textbf{CORS Configuration:} Whitelist specific origins, not \texttt{*}
    \item \textbf{Rate Limiting:} Protect authentication endpoints (5 attempts/15 min)
    \item \textbf{Input Validation:} Use Joi or similar for schema validation
    \item \textbf{Secrets Management:} Store credentials in environment variables
    \item \textbf{HTTPS Only:} Enforce SSL/TLS in production
    \item \textbf{Security Headers:} Include HSTS, X-Frame-Options, CSP
    \item \textbf{Audit Logging:} Log all authentication events
\end{enumerate}

\subsection{Database Security}

\begin{enumerate}
    \item \textbf{Connection Security:} Use SSL/TLS for database connections
    \item \textbf{Authentication:} Enforce strong credentials, disable default accounts
    \item \textbf{Access Control:} Use principle of least privilege for database users
    \item \textbf{Encryption:} Enable encryption at rest and in transit
    \item \textbf{Backup Strategy:} Automated daily backups with tested recovery
    \item \textbf{Monitoring:} Alert on unauthorized access attempts
    \item \textbf{Query Injection:} Use parameterized queries/ORM exclusively
    \item \textbf{Data Masking:} Mask sensitive data in logs and monitoring
\end{enumerate}

\section{Troubleshooting Common Issues}

\subsection{Docker Network Issues}

\begin{itemize}
    \item \textbf{Problem:} Containers cannot communicate
    \item \textbf{Solution:} Use custom bridge network, reference containers by service name
    \item \textbf{Debugging:} Use \texttt{docker network ls}, \texttt{docker inspect <network>}
\end{itemize}

\subsection{Lambda Timeout}

\begin{itemize}
    \item \textbf{Problem:} Lambda function exceeds timeout limit
    \item \textbf{Solution:} Increase timeout in configuration, optimize code, use SNS/SQS for async
    \item \textbf{Monitoring:} Use CloudWatch Logs to track execution time
\end{itemize}

\subsection{DynamoDB Throttling}

\begin{itemize}
    \item \textbf{Problem:} Requests get throttled, high latency
    \item \textbf{Solution:} Increase provisioned capacity or enable auto-scaling
    \item \textbf{Alternative:} Implement exponential backoff retry logic
\end{itemize}

\subsection{Socket.IO Connection Issues}

\begin{itemize}
    \item \textbf{Problem:} WebSocket connection fails
    \item \textbf{Solution:} Check CORS configuration, verify firewall rules
    \item \textbf{Fallback:} Enable HTTP long-polling as fallback transport
\end{itemize}
